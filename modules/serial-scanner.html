async function startCamera() { if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) { statusText.textContent = 'Camera API unavailable (requires HTTPS or iframe allow="camera"). Use "Choose File" below.'; captureBtn.disabled = true; return; } // Try rear camera first, then fall back to any available camera (e.g., laptop webcam) const constraintOptions = [ { video: { facingMode: { ideal: 'environment' } }, audio: false }, { video: true, audio: false } ]; for (const constraints of constraintOptions) { try { stream = await navigator.mediaDevices.getUserMedia(constraints); video.srcObject = stream; await video.play(); statusText.textContent = 'Camera ready.'; captureBtn.disabled = false; return; } catch (err) { console.warn('Camera attempt failed with constraints:', constraints, err.name); statusText.textContent = `Camera error (${err.name}): ${err.message}. Use "Choose File" below.`; } } captureBtn.disabled = true; }
